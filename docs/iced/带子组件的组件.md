# 带子组件的组件

当我们使用 [advanced](https://docs.rs/iced/0.13.1/iced/advanced/widget/trait.Widget.html) 模块创建自定义组件时，我们可以让组件包含子组件。

为了包含子组件，我们需要在结构体中存储子组件。

```rust
struct MyWidget {
    child: Element,
}
```

然后，在 [draw](https://docs.rs/iced/0.13.1/iced/advanced/widget/trait.Widget.html#tymethod.draw) 方法中绘制子组件。

```rust
fn draw(
    &self,
    state: &Tree,
    renderer: &mut Renderer,
    theme: &Theme,
    style: &renderer::Style,
    layout: Layout<'_>,
    cursor: mouse::Cursor,
    viewport: &Rectangle,
) {
    let child_layout = layout.children().next().unwrap();
    let child_state = &state.children[0];

    self.child.as_widget().draw(
        self.child.as_widget(),
        child_state,
        renderer,
        theme,
        style,
        child_layout,
        cursor,
        viewport,
    );
}
```

完整代码如下：

```rust
use iced::{
    Border, Color, Element, Length, Rectangle, Shadow, Size, Theme,
    advanced::{
        Layout, Widget, layout, mouse,
        renderer::{self, Quad},
        widget::Tree,
    },
    widget::{button, container},
};

fn main() -> iced::Result {
    iced::run("My App", MyApp::update, MyApp::view)
}

#[derive(Debug, Clone)]
enum Message {}

#[derive(Default)]
struct MyApp;

impl MyApp {
    fn update(&mut self, _message: Message) {}

    fn view(&self) -> iced::Element<Message> {
        container(MyWidget::with_child(button("Click me!").into()))
            .width(Length::Fill)
            .height(Length::Fill)
            .center_x(Length::Fill)
            .center_y(Length::Fill)
            .into()
    }
}

struct MyWidget {
    child: Element,
}

impl MyWidget {
    fn with_child(child: Element) -> Self {
        Self { child }
    }
}

impl<Message, Renderer> Widget<Message, Theme, Renderer> for MyWidget
where
    Renderer: iced::advanced::Renderer,
{
    fn size(&self) -> Size<Length> {
        Size {
            width: Length::Shrink,
            height: Length::Shrink,
        }
    }

    fn layout(
        &self,
        tree: &mut Tree,
        renderer: &Renderer,
        limits: &layout::Limits,
    ) -> layout::Node {
        let child = tree.children[0];
        let child_layout = child.as_widget().layout(child, renderer, limits);

        let size = limits.resolve(child_layout.size());

        layout::Node::with_children(size, vec![child_layout])
    }

    fn draw(
        &self,
        state: &Tree,
        renderer: &mut Renderer,
        theme: &Theme,
        style: &renderer::Style,
        layout: Layout<'_>,
        cursor: mouse::Cursor,
        viewport: &Rectangle,
    ) {
        let child_layout = layout.children().next().unwrap();
        let child_state = &state.children[0];

        renderer.fill_quad(
            Quad {
                bounds: layout.bounds(),
                border: Border {
                    color: Color::from_rgb(0.4, 0.4, 1.0),
                    width: 1.0,
                    radius: 10.0.into(),
                },
                shadow: Shadow::default(),
            },
            Color::from_rgb(0.6, 0.6, 1.0),
        );

        self.child.as_widget().draw(
            self.child.as_widget(),
            child_state,
            renderer,
            theme,
            style,
            child_layout,
            cursor,
            viewport,
        );
    }

    fn children(&self) -> Vec<&Tree> {
        vec![self.child.as_widget().tree()]
    }

    fn diff(&self, tree: &mut Tree) {
        tree.diff_children(std::slice::from_ref(&self.child));
    }
}

impl<'a, Message, Renderer> From<MyWidget> for Element<'a, Message, Theme, Renderer>
where
    Renderer: iced::advanced::Renderer,
{
    fn from(widget: MyWidget) -> Self {
        Self::new(widget)
    }
}
```
