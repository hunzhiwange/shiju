# 接受任意子组件

当我们使用 [advanced](https://docs.rs/iced/0.13.1/iced/advanced/widget/trait.Widget.html) 模块创建自定义组件时，我们可以让组件接受任意子组件。

为了接受子组件，我们需要在结构体中存储一个 `Vec<Element>` 字段。

```rust
struct MyWidget {
    children: Vec<Element>,
}
```

然后在 [layout](https://docs.rs/iced/0.13.1/iced/advanced/widget/trait.Widget.html#tymethod.layout) 方法中计算子组件的布局。

```rust
fn layout(
    &self,
    tree: &mut Tree,
    renderer: &Renderer,
    limits: &layout::Limits,
) -> layout::Node {
    let children = tree.children[..self.children.len()]
        .iter_mut()
        .zip(self.children.iter())
        .map(|(state, child)| {
            child.as_widget().layout(child, state, renderer, limits)
        })
        .collect::<Vec<_>>();

    layout::Node::with_children(limits.resolve(), children)
}
```

最后，在 [draw](https://docs.rs/iced/0.13.1/iced/advanced/widget/trait.Widget.html#tymethod.draw) 方法中绘制子组件。

```rust
fn draw(
    &self,
    state: &Tree,
    renderer: &mut Renderer,
    theme: &Theme,
    style: &renderer::Style,
    layout: Layout<'_>,
    cursor: mouse::Cursor,
    viewport: &Rectangle,
) {
    for ((child_state, child), layout) in state.children[..self.children.len()]
        .iter()
        .zip(&self.children)
        .zip(layout.children())
    {
        child
            .as_widget()
            .draw(child, child_state, renderer, theme, style, layout, cursor, viewport);
    }
}
```

完整代码如下：

```rust
use iced::{
    Border, Color, Element, Length, Rectangle, Shadow, Size, Theme,
    advanced::{
        Layout, Widget, layout, mouse,
        renderer::{self, Quad},
        widget::Tree,
    },
    widget::container,
};

fn main() -> iced::Result {
    iced::run("My App", MyApp::update, MyApp::view)
}

#[derive(Debug, Clone)]
enum Message {}

#[derive(Default)]
struct MyApp;

impl MyApp {
    fn update(&mut self, _message: Message) {}

    fn view(&self) -> iced::Element<Message> {
        container(MyWidget::with_children(vec!["Child 1".into(), "Child 2".into()]))
            .width(Length::Fill)
            .height(Length::Fill)
            .center_x(Length::Fill)
            .center_y(Length::Fill)
            .into()
    }
}

struct MyWidget {
    children: Vec<Element>,
}

impl MyWidget {
    fn with_children(children: Vec<Element>) -> Self {
        Self { children }
    }
}

impl<Message, Renderer> Widget<Message, Theme, Renderer> for MyWidget
where
    Renderer: iced::advanced::Renderer,
{
    fn size(&self) -> Size<Length> {
        Size {
            width: Length::Shrink,
            height: Length::Shrink,
        }
    }

    fn layout(
        &self,
        tree: &mut Tree,
        renderer: &Renderer,
        limits: &layout::Limits,
    ) -> layout::Node {
        let children = tree.children[..self.children.len()]
            .iter_mut()
            .zip(self.children.iter())
            .map(|(state, child)| {
                child.as_widget().layout(child, state, renderer, limits)
            })
            .collect::<Vec<_>>();

        layout::Node::with_children(limits.resolve(), children)
    }

    fn draw(
        &self,
        state: &Tree,
        renderer: &mut Renderer,
        theme: &Theme,
        style: &renderer::Style,
        layout: Layout<'_>,
        cursor: mouse::Cursor,
        viewport: &Rectangle,
    ) {
        for ((child_state, child), layout) in state.children[..self.children.len()]
            .iter()
            .zip(&self.children)
            .zip(layout.children())
        {
            child.as_widget().draw(
                child,
                child_state,
                renderer,
                theme,
                style,
                layout,
                cursor,
                viewport,
            );
        }
    }

    fn children(&self) -> Vec<&Tree> {
        self.children.iter().map(|e| e.as_widget().tree()).collect()
    }

    fn diff(&self, tree: &mut Tree) {
        tree.diff_children(&self.children);
    }
}

impl<'a, Message, Renderer> From<MyWidget> for Element<'a, Message, Theme, Renderer>
where
    Renderer: iced::advanced::Renderer,
{
    fn from(widget: MyWidget) -> Self {
        Self::new(widget)
    }
}
```
