
#[test]
#[ignore = "explicit task 91 native macOS UI acceptance"]
fn cc_base_icon_task91_ui_server() -> anyhow::Result<()> {
    use axum::{body::Body, http::{Request, StatusCode}, middleware::{Next, from_fn}, response::IntoResponse};
    let root = std::path::PathBuf::from(std::env::var("TASK91_UI_ROOT")?);
    tokio::runtime::Runtime::new()?.block_on(async {
        let db = GlobalDatabase::for_home(&root.join("db"))?;
        let seed = Server::start(db.clone(), None).await?;
        seed.client("task91-platform", "task91-company", true)?.custom_icon_add(&AddCustomIconRequest { name: "工作图标·中文验收".into(), data: PNG.into() }).await.map_err(anyhow::Error::msg)?;
        seed.stop().await?;
        let key = uuid::Uuid::new_v4().to_string();
        let guard = crate::security::pairing::PairingGuard::new(true, std::slice::from_ref(&key));
        let listener = tokio::net::TcpListener::bind(("127.0.0.1", 0)).await?;
        let addr = listener.local_addr()?;
        std::fs::write(root.join("gateway-client-bootstrap.json"), serde_json::to_vec(&serde_json::json!({
            "host":"127.0.0.1", "port":addr.port(), "platform_id":"task91-platform", "company_id":"task91-company", "skey":key,
            "active_server_id":"task91", "servers":[{"id":"task91", "name":"本地图标验收91", "host":"127.0.0.1", "port":addr.port(), "platform_id":"task91-platform", "company_id":"task91-company", "skey":key}]
        }))?)?;
        let counter = std::sync::Arc::new(std::sync::atomic::AtomicUsize::new(0));
        let middleware_root = root.clone();
        let app = crate::gateway::router::build_router(Vec::new()).layer(from_fn(move |req: Request<Body>, next: Next| {
            let guard = guard.clone(); let root = middleware_root.clone(); let counter = counter.clone();
            async move {
                if !guard.is_authenticated(crate::gateway::api::auth::extract_auth_skey(req.headers()).unwrap_or("")) { return StatusCode::UNAUTHORIZED.into_response(); }
                if req.method() == axum::http::Method::PATCH && req.uri().path().starts_with("/v1/agent-apps/custom-icons/") {
                    let count = counter.fetch_add(1, std::sync::atomic::Ordering::SeqCst) + 1;
                    std::fs::write(root.join("patch-count"), count.to_string()).unwrap();
                    let deadline = std::time::Instant::now() + std::time::Duration::from_secs(120);
                    while root.join("hold").exists() && std::time::Instant::now() < deadline { tokio::time::sleep(std::time::Duration::from_millis(100)).await; }
                }
                next.run(req).await
            }
        })).layer(axum::Extension(db));
        let mut listener = Some(listener);
        for _ in 0..12 {
            let current = match listener.take() { Some(value) => value, None => tokio::net::TcpListener::bind(addr).await? };
            let (send, receive) = tokio::sync::oneshot::channel();
            let router = app.clone();
            let task = tokio::spawn(async move { axum::serve(current, router).with_graceful_shutdown(async { let _ = receive.await; }).await });
            std::fs::write(root.join("ready"), format!("http://{addr}"))?;
            let deadline = std::time::Instant::now() + std::time::Duration::from_secs(1800);
            while !root.join("offline").exists() && !root.join("stop").exists() && std::time::Instant::now() < deadline { tokio::time::sleep(std::time::Duration::from_millis(100)).await; }
            let _ = send.send(()); task.await??;
            std::fs::remove_file(root.join("ready"))?;
            std::fs::write(root.join("offline-confirmed"), "listener stopped")?;
            if root.join("stop").exists() { return Ok(()); }
            while root.join("offline").exists() && !root.join("stop").exists() { tokio::time::sleep(std::time::Duration::from_millis(100)).await; }
            if root.join("stop").exists() { return Ok(()); }
            std::fs::remove_file(root.join("offline-confirmed"))?;
        }
        anyhow::bail!("native acceptance restart budget exhausted")
    })
}
